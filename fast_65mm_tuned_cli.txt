# Betaflight CLI configuration for BetaFPV G473 1S Air 65
# start the command batch
batch start

# reset configuration to default settings
defaults nosave

# board metadata
board_name BETAFPVG473
manufacturer_id BEFH

# craft name
set craft_name = "1S Air 65"
set motor_kv = 23000

# filtering tuned for ~800 Hz hover (~48k RPM) on 3-blade props
set gyro_lpf1_static_hz = 150
set gyro_lpf2_static_hz = 120
set dyn_notch_count = 1
set dyn_notch_min_hz = 250
set dyn_notch_max_hz = 1000
set dyn_notch_q = 200
set rpm_filter_min_hz = 500
set rpm_filter_fade_range_hz = 200
set rpm_filter_lpf_hz = 300

# throttle
set throttle_limit_type = SCALE
set throttle_limit_percent = 100

# AUX2 three position switch: Angle -> Horizon -> AirMode with dedicated rate profiles
aux 0 1 2 900 1300 0 0    # ANGLE + RateProfile0
aux 1 2 2 1301 1700 0 0   # HORIZON + RateProfile1
aux 2 23 2 1701 2100 0 0  # AIRMODE + RateProfile2

# PID and Feed Forward values per profile
profile 0
set p_pitch = 50
set p_roll = 50
set i_pitch = 70
set i_roll = 70
set d_pitch = 45
set d_roll = 45
set f_pitch = 50
set f_roll = 50

profile 1
set p_pitch = 65
set p_roll = 65
set i_pitch = 80
set i_roll = 80
set d_pitch = 55
set d_roll = 55
set f_pitch = 85
set f_roll = 85

profile 2
set p_pitch = 75
set p_roll = 75
set i_pitch = 90
set i_roll = 90
set d_pitch = 60
set d_roll = 60
set f_pitch = 100
set f_roll = 100

# RC rates/expo per rate profile
rateprofile 0
set roll_rc_rate = 110
set pitch_rc_rate = 110
set yaw_rc_rate = 110
set roll_expo = 25
set pitch_expo = 25
set yaw_expo = 20

rateprofile 1
set roll_rc_rate = 110
set pitch_rc_rate = 110
set yaw_rc_rate = 110
set roll_expo = 25
set pitch_expo = 25
set yaw_expo = 20

rateprofile 2
set roll_rc_rate = 120
set pitch_rc_rate = 120
set yaw_rc_rate = 110
set roll_expo = 15
set pitch_expo = 15
set yaw_expo = 10

# restore default profile selections
profile 0
rateprofile 0

# save configuration
save
